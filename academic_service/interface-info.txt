Auth("POST", "/copilot_for_docs/doc_fulltext", request.GetDevice().GetXDeviceId(), c)
func (o *Auth) Auth(method, url string, deviceId string, c *gin.Context) bool {
	timestamp := c.GetHeader("timestamp")
	token := c.GetHeader("token")
	return CheckSignKgs(token, method, url, deviceId, timestamp, o.key)
}
func CheckSignKgs(token, method, url, deviceId, timestamp string, key []byte) bool {
	if ts, err := strconv.ParseInt(timestamp, 10, 64); err != nil {
		return false
	} else {
		tsn := time.Now().UnixNano() / 1e6
		limitMilliSecond := float64(600000)
		if math.Abs(float64(tsn-ts)) > limitMilliSecond {
			return false
		}
	}
	data := method + "&" + url + "&deviceId=" + deviceId + "&timestamp=" + timestamp
	signature := hmac.New(sha256.New, key)
	if _, err := signature.Write([]byte(data)); err != nil {
		logs.Error(err.Error())
	}
	if base64.StdEncoding.EncodeToString(signature.Sum(nil)) == token {
		return true
	}
	return false
}
